(async () => { if (!navigator.xr) { console.log("❌ WebXR not available at all"); return; } // First check if AR sessions in general are supported const arOK = await navigator.xr.isSessionSupported('immersive-ar'); console.log("WebXR AR supported?", arOK); // Now check specifically for image-tracking const imageOK = await navigator.xr.isSessionSupported('immersive-ar', { requiredFeatures: ['image-tracking'] }); console.log("WebXR Image Tracking supported?", imageOK); })();
