async function init() { setStatus('🔍 Checking WebXR…'); if (!navigator.xr) return fail('WebXR not available'); setStatus('✅ WebXR available'); const arOK = await navigator.xr.isSessionSupported('immersive-ar'); if (!arOK) return fail('AR not supported'); setStatus('✅ AR supported'); // 1) Load marker.png setStatus('📥 Loading marker.png…'); let bitmap; try { const img = new Image(); img.src = 'marker.png'; await img.decode(); bitmap = await createImageBitmap(img); setStatus('✅ marker.png loaded'); } catch (e) { console.error(e); return fail('❌ marker.png failed to load'); } // 2) Three.js + ARButton scene = new THREE.Scene(); camera = new THREE.PerspectiveCamera(); renderer = new THREE.WebGLRenderer({ canvas: document.getElementById('xr-canvas'), alpha: true, antialias: true }); renderer.xr.enabled = true; renderer.setSize(window.innerWidth, window.innerHeight); document.body.appendChild( ARButton.createButton(renderer, { requiredFeatures: ['local-floor','image-tracking'], trackedImages: [{ image: bitmap, widthInMeters: 0.2 }] }) ); // 3) Light + hidden box scene.add(new THREE.HemisphereLight(0xffffff, 0xbbbbff, 1)); box = new THREE.Mesh( new THREE.BoxGeometry(0.2,0.2,0.2), new THREE.MeshStandardMaterial({ color: 0x00ff00 }) ); box.visible = false; scene.add(box); // 4) On session start, get ref space renderer.xr.addEventListener('sessionstart', async () => { xrRefSpace = await renderer.xr.getSession() .requestReferenceSpace('local-floor'); setStatus('▶️ Point at marker.png...'); }); // 5) Render loop renderer.setAnimationLoop(onXRFrame); } async function onXRFrame(time, frame) { // 6) If not anchored yet and not locked, track image if (!xrAnchor && !anchorLocked) { for (const result of frame.getImageTrackingResults()) { if (result.trackingState === 'tracked') { setStatus('📌 Image detected! Creating anchor…'); const pose = frame.getPose(result.imageSpace, xrRefSpace); if (pose) { xrAnchor = await frame.createAnchor(pose.transform, xrRefSpace); box.visible = true; setStatus('✅ Anchor created, box placed'); // after 3s, lock anchor to ignore further tracking setTimeout(() => { anchorLocked = true; setStatus('🔒 Anchor locked in world space'); }, 3000); } break; } } } // 7) If anchored, update box from anchor if (xrAnchor) { const anchorPose = frame.getPose(xrAnchor.anchorSpace, xrRefSpace); if (anchorPose) { box.matrix.fromArray(anchorPose.transform.matrix); box.matrix.decompose(box.position, box.quaternion, box.scale); box.matrixAutoUpdate = false; } } // 8) Always render renderer.render(scene, camera); } function fail(msg) { setStatus(msg); }
